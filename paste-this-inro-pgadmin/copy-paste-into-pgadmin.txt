-- ================= ADMINS =================

CREATE TABLE admins (
    id            INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    admin_number  VARCHAR(20)  NOT NULL UNIQUE,   -- what admins type to log in, e.g. A-001
    password_hash VARCHAR(255) NOT NULL
);

CREATE TABLE admin_details (
    id         INTEGER PRIMARY KEY REFERENCES admins(id) ON DELETE CASCADE,
    username   VARCHAR(30)  NOT NULL UNIQUE,
    email      VARCHAR(255) NOT NULL UNIQUE,
    phone      VARCHAR(15),
    created_at TIMESTAMP    NOT NULL DEFAULT now()
);

CREATE TABLE admin_profiles (
    id               INTEGER PRIMARY KEY REFERENCES admins(id) ON DELETE CASCADE,
    bio              VARCHAR(280) NOT NULL DEFAULT '',
    color            VARCHAR(7)   NOT NULL DEFAULT '#1E3A8A',
    use_gradient     BOOLEAN      NOT NULL DEFAULT false,
    gradient_color_2 VARCHAR(7),
    text_color       VARCHAR(7)   NOT NULL DEFAULT '#FFFFFF'
);


-- ================= STUDENTS =================

CREATE TABLE users (
    id             INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    student_number VARCHAR(20)  NOT NULL UNIQUE,  -- what students type to log in, e.g. 2023-00123
    password_hash  VARCHAR(255) NOT NULL,
    banned         BOOLEAN      NOT NULL DEFAULT false,
    banned_at      TIMESTAMP,
    banned_by      INTEGER REFERENCES admins(id) ON DELETE SET NULL
);

CREATE TABLE user_details (
    id         INTEGER PRIMARY KEY REFERENCES users(id) ON DELETE CASCADE,
    username   VARCHAR(30)  NOT NULL UNIQUE,
    email      VARCHAR(255) NOT NULL UNIQUE,
    phone      VARCHAR(15),
    created_at TIMESTAMP    NOT NULL DEFAULT now()
);

CREATE TABLE user_profiles (
    id                  INTEGER PRIMARY KEY REFERENCES users(id) ON DELETE CASCADE,
    bio                 VARCHAR(280) NOT NULL DEFAULT '',
    pfp                 VARCHAR(255),                       -- file path, not the image itself
    color               VARCHAR(7)   NOT NULL DEFAULT '#1E3A8A',
    use_gradient        BOOLEAN      NOT NULL DEFAULT false,
    gradient_color_2    VARCHAR(7),
    text_color          VARCHAR(7)   NOT NULL DEFAULT '#FFFFFF',
    show_borrowed_books BOOLEAN      NOT NULL DEFAULT true
);


-- ================= CATALOG =================

CREATE TABLE books (
    id               INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    title            VARCHAR(255) NOT NULL,
    authors          TEXT         NOT NULL,           -- comma-separated
    media_type       VARCHAR(20)  NOT NULL DEFAULT 'book'
                     CHECK (media_type IN ('book', 'audiovisual', 'board_game', 'digital_media')),
    isbn             VARCHAR(20),
    published_date   DATE,
    classification   VARCHAR(255),
    key_word         TEXT,                            -- comma-separated tags
    description      TEXT,
    shelf_number     VARCHAR(20),
    cover_image      TEXT,                            -- uploaded path or image URL
    ebook_path       VARCHAR(255),
    total_copies     INTEGER      NOT NULL DEFAULT 1 CHECK (total_copies >= 0),
    copies_available INTEGER      NOT NULL DEFAULT 1 CHECK (copies_available >= 0),
    added_on         TIMESTAMP    NOT NULL DEFAULT now()
    -- Add CHECK (copies_available <= total_copies) once the admin "add stock"
    -- code updates both columns.
);


-- ================= LISTS =================

CREATE TABLE lists (
    id         INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    user_id    INTEGER      NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    name       VARCHAR(100) NOT NULL,
    created_at TIMESTAMP    NOT NULL DEFAULT now()
);

CREATE TABLE list_books (
    list_id  INTEGER   NOT NULL REFERENCES lists(id) ON DELETE CASCADE,
    book_id  INTEGER   NOT NULL REFERENCES books(id) ON DELETE CASCADE,
    added_at TIMESTAMP NOT NULL DEFAULT now(),
    PRIMARY KEY (list_id, book_id)                    -- a book can't be in a list twice
);


-- ================= BORROWING =================

CREATE TABLE borrowed_books (
    id          INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    user_id     INTEGER     NOT NULL REFERENCES users(id),
    book_id     INTEGER     NOT NULL REFERENCES books(id),
    status      VARCHAR(10) NOT NULL DEFAULT 'pending'
                CHECK (status IN ('pending', 'approved', 'rejected')),
    borrowed_at TIMESTAMP   NOT NULL DEFAULT now(),   -- when the request was made
    approved_at TIMESTAMP,
    approved_by INTEGER REFERENCES admins(id) ON DELETE SET NULL,
    due_date    DATE,
    returned_at TIMESTAMP
);

CREATE INDEX idx_borrowed_books_user ON borrowed_books (user_id);
CREATE INDEX idx_borrowed_books_book ON borrowed_books (book_id);

-- One open request or loan per student per book
CREATE UNIQUE INDEX uniq_open_borrow ON borrowed_books (user_id, book_id)
    WHERE returned_at IS NULL AND status IN ('pending', 'approved');
